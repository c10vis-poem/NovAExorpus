## Part 1: The Open Wiki / Reasoning Bank Bridge Parser Script
This functional script serves as the core automation component for Priority Phase 3 & 4. It runs locally inside your Termux environment.
It acts as the functional engine that parses incoming JSON-RPC traffic from your Open Wiki CLI, verifies the requested parameters against your JSON-RPC schemas, updates or reviews the state strings inside the Reasoning Bank Ledger, and appends clean, audited entries down into your Obsidian Vault markdown files.

#!/usr/bin/env python3"""
AESOP XI: COGNITIVE MIDDLEWARE PARSER & EXECUTION BRIDGE
Validates Open Wiki CLI JSON-RPC tokens, manages the local Reasoning Bank Ledger state,
and writes human-approved markdown payloads safely down into your Obsidian Vault repository layers."""
import osimport jsonimport sys
LEDGER_PATH = "../../file-management-and-skills/reasoning-bank-ledger/active_execution_paths.json"OBSIDIAN_VAULT_PATH = "../../obsidian-vault-new"BLUEPRINT_INDEX = "../../master_blueprint.txt"
def load_json_file(file_path):
    """Safely handles reading local JSON databases with schema isolation fallbacks."""
    if not os.path.exists(file_path):
        return {}
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            return json.load(f)
    except json.JSONDecodeError:
        print(f"❌ DATABASE ERROR: Structural parsing failure inside file: {file_path}")
        return {}
def save_json_file(file_path, data_payload):
    """Enforces zero-trust transactional saves back to your hardware partitions."""
    with open(file_path, "w", encoding="utf-8") as f:
        json.dump(data_payload, f, indent=4)
def execute_json_rpc_command(raw_rpc_string):
    print("⏳ [WIKI BRIDGE] Intercepting incoming JSON-RPC command array...")
    
    # 1. Structural Validation Checklist Check
    try:
        rpc = json.loads(raw_rpc_string)
    except json.JSONDecodeError:
        return {"jsonrpc": "2.0", "error": {"code": -32700, "message": "Parse error: Invalid JSON syntax payload."}, "id": None}

    # Enforce basic standard JSON-RPC structural parameter checks
    if "jsonrpc" not in rpc or "method" not in rpc or "params" not in rpc or "id" not in rpc:
        return {"jsonrpc": "2.0", "error": {"code": -32600, "message": "Invalid Request: Schema mapping attributes missing."}, "id": rpc.get("id")}

    method = rpc["method"]
    params = rpc["params"]
    session_id = params.get("associated_session_id")
    request_id = rpc["id"]

    if not session_id:
        return {"jsonrpc": "2.0", "error": {"code": -32602, "message": "Invalid Params: 'associated_session_id' parameter field is mandatory."}, "id": request_id}

    # Load active data arrays from our persistent layers
    ledger = load_json_file(LEDGER_PATH)
    if "active_sessions" not in ledger:
        ledger["active_sessions"] = {}

    # --------------------------------------------------------------------------
    # METHOD A: WIKI.REASONING_BANK.COMMIT_STATE
    # --------------------------------------------------------------------------
    if method == "wiki.reasoning_bank.commit_state":
        print(f"🧠 [LEDGER EXECUTOR] Syncing active mental chains for task session: {session_id}")
        
        # Pull payload metrics from input parameter keys
        thought_log = params.get("payload_content_markdown", "No thought trace supplied.")
        tool_intent = params.get("target_file_name", "None Specified")
        
        # Instantiate or append data strings straight to our session profile records
        if session_id not in ledger["active_sessions"]:
            ledger["active_sessions"][session_id] = {
                "lifecycle_state": "ACTIVE_INFERENCE_RUN",
                "chain_of_thought_stack": [],
                "hardware_telemetry_snapshot": {"npu_offload_status": "HTP_PATHWAY_ACTIVE"}
            }
            
        step_num = len(ledger["active_sessions"][session_id]["chain_of_thought_stack"]) + 1
        ledger["active_sessions"][session_id]["chain_of_thought_stack"].append({
            "step_sequence_id": step_num,
            "agent_node_id": "TERMUX_CLI_MIDDLEWARE",
            "thought_process_log": thought_log,
            "tool_intent_selected": tool_intent
        })
        
        save_json_file(LEDGER_PATH, ledger)
        print(f"✅ Success: Step {step_num} successfully pinned to local memory registries.")
        return {"jsonrpc": "2.0", "result": {"status": "SUCCESS_STATE_COMMITTED", "total_steps_logged": step_num}, "id": request_id}

    # --------------------------------------------------------------------------
    # METHOD B: WIKI.NOTE.APPEND_SANITIZED
    # --------------------------------------------------------------------------
    elif method == "wiki.note.append_sanitized":
        print("💎 [VAULT EXECUTOR] Initializing local Obsidian Markdown note mutation pass...")
        
        target_note = params.get("target_file_name")
        markdown_data = params.get("payload_content_markdown")
        human_verified = params.get("human_verification_flag_attached", False)

        if not target_note or not markdown_data:
            return {"jsonrpc": "2.0", "error": {"code": -32602, "message": "Invalid Params: Note filename and content strings are mandatory values."}, "id": request_id}

        # Enforce your Zero-Trust air-gap law explicitly at compile boundary lines
        if not human_verified:
            print("❌ SECURITY REJECTION: Data payload modification blocked. Human Air-Gap verification tag is missing.")
            return {"jsonrpc": "2.0", "error": {"code": -32603, "message": "Internal Error: Human-In-The-Loop air-gap validation token required to write code files."}, "id": request_id}

        # Build paths securely matching your strict repository directories mapping
        note_output_path = os.path.join(OBSIDIAN_VAULT_PATH, target_note)
        
        print(f"📝 Writing sanitized documentation string block straight into vault link: {note_output_path}")
        with open(note_output_path, "a", encoding="utf-8") as vault_f:
            vault_f.write(f"\n\n## Session System Entry Log [{session_id}]\n{markdown_data}")

        # Update tracking database hashes inside our master registration files indexes
        with open(BLUEPRINT_INDEX, "a", encoding="utf-8") as blueprint_f:
            blueprint_f.write(f"DIRHASH: obsidian-vault-new/{target_note} | VALIDATION: VERIFIED_MARKDOWN_ASSET\n")

        # Advance ledger phase mapping parameters safely
        if session_id in ledger["active_sessions"]:
            ledger["active_sessions"][session_id]["lifecycle_state"] = "COMMITTED_AND_CLOSED"
            save_json_file(LEDGER_PATH, ledger)

        print("✅ Success: Vault modified. Hash configuration targets tracked cleanly.")
        return {"jsonrpc": "2.0", "result": {"status": "MARKDOWN_NOTE_APPENDED_SUCCESSFULLY", "file_mutated": target_note}, "id": request_id}

    else:
        return {"jsonrpc": "2.0", "error": {"code": -32601, "message": f"Method not found: '{method}' is not registered inside the network controller mapping matrix."}, "id": request_id}
if __name__ == "__main__":
    # Test Ingress Mock Data Verification Simulation Loop
