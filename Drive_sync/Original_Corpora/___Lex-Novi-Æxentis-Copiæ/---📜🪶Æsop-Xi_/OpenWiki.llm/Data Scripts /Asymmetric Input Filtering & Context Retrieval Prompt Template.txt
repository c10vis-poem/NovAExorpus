## Part 2: Automated JSONL Extraction & Context Router Engine
To implement the tracking logic defined in the prompt above, save this Python processing automation script inside your Termux environment as file-management-and-skills/omni_route_extractor.py.
This code runs locally to scan user goals, execute the regex extraction rules against your compiled .jsonl skill files, and intelligently switch paths to raw text files or external web searches depending on your exact operational context.

#!/usr/bin/env python3"""
AESOP XI: OMNI ROUTE EXTRACTOR & CONTEXT ENGINE
Handles asymmetric filtering rules. Decides when to stream structured .jsonl rows
vs parsing un-quantized .txt documentation, and manages web view tool verification hooks."""
import osimport jsonimport reimport sys
JSONL_SKILLS_PATH = "skill-construction-factory/compiled_capabilities_training.jsonl"RAW_DOCS_DIR = "target-docs-curation/"
def analyze_and_route_context(user_raw_input):
    print("⏳ [OMNI ROUTE] Evaluating raw ingress input strings...")
    
    # 1. Clean the input telemetry against basic script parameters
    sanitized_input = user_raw_input.strip()
    
    # 2. Rule Check 1: Detect explicit tool execution indicators (Command/Code request)
    is_execution_task = any(keyword in sanitized_input.lower() for keyword in ["run", "execute", "build", "bash", "script", "terminal", "compile"])
    
    # 3. Rule Check 2: Detect deep analysis indicators
    is_deep_analysis = any(keyword in sanitized_input.lower() for keyword in ["error", "mismatch", "debug", "kernel", "register", "equation", "why"])

    # 4. Rule Check 3: Live online variable lookups
    requires_web_check = any(keyword in sanitized_input.lower() for keyword in ["search", "latest change", "current api", "online doc", "verify online"])

    # --------------------------------------------------------------------------
    # EXECUTION PATHWAY A: STRUCTURED JSONL DATASET EXTRACTION
    # --------------------------------------------------------------------------
    if is_execution_task and not is_deep_analysis and not requires_web_check:
        print("🎯 ROUTING PATHWAY IDENTIFIED: [COMPILED_JSONL_SKILLS]")
        if not os.path.exists(JSONL_SKILLS_PATH):
            print(f"❌ ROUTING ERROR: Dataset missing at '{JSONL_SKILLS_PATH}'.")
            return None

        print("🗜️ Initializing structural token extraction from dataset rows...")
        extracted_skills = []
        
        # Regex engines matching your string extraction directives
        user_pattern = re.compile(r'"user":\s*"([^"]*)"')
        assistant_pattern = re.compile(r'"assistant":\s*"([^"]*)"')
        
        with open(JSONL_SKILLS_PATH, "r", encoding="utf-8") as f:
            for line in f:
                # Search for target keyword intersections inside the jsonl records
                if any(kw in line.lower() for kw in sanitized_input.split()):
                    user_match = user_pattern.search(line)
                    assistant_match = assistant_pattern.search(line)
                    
                    if user_match and assistant_match:
                        extracted_skills.append({
                            "intent": user_match.group(1),
                            "action_payload": assistant_match.group(1)
                        })
        
        print(f"✅ Extracted {len(extracted_skills)} matching tool skills from dataset layer.")
        return {"route": "JSONL_SKILLS", "payload": extracted_skills}

    # --------------------------------------------------------------------------
    # EXECUTION PATHWAY B: RAW TEXT PARSING OVER EVERY SINGLE WORD
    # --------------------------------------------------------------------------
    elif is_deep_analysis and not requires_web_check:
        print("📖 ROUTING PATHWAY IDENTIFIED: [RAW_TXT_ENCYCLOPEDIA]")
        print("⚡ Bypassing summarized chunks. Scanning raw data libraries line-by-line...")
        
        verbatim_logs = []
        for root, _, files in os.walk(RAW_DOCS_DIR):
            for file in files:
                if file.endswith(".txt") or file.endswith(".md"):
                    file_path = os.path.join(root, file)
                    with open(file_path, "r", encoding="utf-8") as raw_f:
                        for line_num, line in enumerate(raw_f, 1):
                            # Target exact syntax string matches inside raw manuals
                            if any(kw in line.lower() for kw in sanitized_input.split()):
                                verbatim_logs.append(f"[{file} | Line {line_num}]: {line.strip()}")
                                
        print(f"✅ Full-text verification complete. Found {len(verbatim_logs)} structural syntax indicators.")
        return {"route": "RAW_TXT_BLOBS", "payload": verbatim_logs[:10]} # Cap tracking size

    # --------------------------------------------------------------------------
    # EXECUTION PATHWAY C: HORIZONS UI WEBVIEW VERIFICATION HOOKS
    # --------------------------------------------------------------------------
    elif requires_web_check:
        print("🌐 ROUTING PATHWAY IDENTIFIED: [WEB_TOOL_VERIFICATION]")
        print("🚨 Initializing argument validation for Chromium Sandbox container...")
        
        # Build a valid structural payload to pass out to Horizons UI browser services
        web_verification_payload = {
            "jsonrpc": "2.0",
            "method": "horizons.ui.webview.scrape",
            "params": {
                "target_verification_query": sanitized_input,
                "sandbox_isolation_enabled": True,
                "cross_check_vs_nope_data": True
            },
            "id": 1
        }
        return {"route": "WEBVIEW_TOOL_HOOK", "payload": web_verification_payload}

    else:
        print("⚖️ ROUTING PATHWAY IDENTIFIED: DEFAULT MODULAR INFERENCE CONTEXT")
        return {"route": "STANDARD_ROUTING", "payload": sanitized_input}
if __name__ == "__main__":
    # Internal baseline test verification routine
    sample_prompt = "Run a bash script compile layout error tracking tool to verify online changes"
    result = analyze_and_route_context(sample_prompt)
    print(json.dumps(result, indent=2))
