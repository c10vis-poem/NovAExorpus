## 📲 The Secure Brave-to-Termux Mobile Extraction Protocol
This method completely avoids browser security blocks because it relies on Brave's native layout engine to handle the text copying, then uses bare-metal Termux utilities to format and clean the data.
## Step 1: Capture the Text in Brave Beta

   1. Open this conversation thread inside your Brave Beta app.
   2. Long-press any word on the screen to bring up the Android selection handles.
   3. Tap the three-dot menu option on the pop-up toolbar and select Select All.
   4. Tap Copy on the native Android system toolbar. (Brave will now save the entire clean text layout, including all code blocks, straight to your system clipboard).

## Step 2: Deploy the Termux Scraper Sanitizer File
Switch to your Termux app terminal window. We will build a native script named harvest_thread.py inside your termux-building-skills/open-wiki-cli/ folder. This script pulls the raw text from your Android clipboard, strips out all tracking telemetry, buttons, site navigation, and interface junk, and formats it cleanly for your AI query layers.
Run this code block to build and register the utility:

#!/usr/bin/env python3"""
AESOP XI: BARE-METAL MOBILE CLIPBOARD SANITIZER
Pulls raw clipboard text from Android via Termux API hooks.
Strips interface data and saves clean markdown histories for memory injections."""
import osimport subprocessimport reimport sys
OUTPUT_PATH = "../../file-management-and-skills/target-docs-curation/unassigned-telemetry/harvested_thread_context.txt"
def get_android_clipboard():
    """Calls Termux-API tools to pull raw text from the Android system clipboard."""
    try:
        # Executes the core termux-clipboard-get command securely
        result = subprocess.run(["termux-clipboard-get"], capture_output=True, text=True, check=True)
        return result.stdout
    except FileNotFoundError:
        print("❌ CRITICAL ERROR: 'termux-api' packages or dependencies are missing inside Termux.")
        print("👉 Action Required: Run 'pkg install termux-api -y' in your terminal screen first.")
        sys.exit(1)
def sanitize_and_save_payload():
    print("⏳ [TERMUX HARVESTER] Intercepting Android clipboard stream...")
    raw_text = get_android_clipboard()

    if not raw_text.strip():
        print("⚠️  CLIPBOARD EMPTY: Copy the Brave Beta thread context before executing.")
        return

    # Structural sanitization: Strip UI noise, duplicate spaces, and broken web lines
    lines = raw_text.split('\n')
    clean_lines = []
    
    # Filter strings to strip button elements, avatar tokens, and UI layout junk
    ignore_patterns = [
        r"^Copy code$", r"^Share$", r"^Regenerate$", r"^Good response$", r"^Bad response$",
        r"^Listen$", r"^View sources$", r"^Edit prompt$", r"^Type a message\.\.\.$"
    ]
    
    for line in lines:
        trimmed = line.strip()
        if not trimmed:
            clean_lines.append("")
            continue
        # Skip UI noise lines
        if any(re.match(pattern, trimmed) for pattern in ignore_patterns):
            continue
        clean_lines.append(trimmed)

    # Collapse consecutive empty vertical padding text blocks
    sanitized_body = "\n".join(clean_lines)
    sanitized_body = re.sub(r'\n{3,}', '\n\n', sanitized_body)

    # Ensure targeted landing directories exist
    os.makedirs(os.path.dirname(OUTPUT_PATH), exist_ok=True)

    with open(OUTPUT_PATH, "w", encoding="utf-8") as f:
        f.write("================================================================================" + "\n")
        f.write("AESOP XI CANONICAL WORKSPACE CONTEXT HARVEST" + "\n")
        f.write("SOURCE INFRASTRUCTURE: Brave Beta Client Node-Alpha Ingress" + "\n")
        f.write("================================================================================" + "\n\n")
        f.write(sanitized_body)

    print(f"🎉 SUCCESS: Telemetry captured. Clean context written to: {OUTPUT_PATH}")
if __name__ == "__main__":
    sanitize_and_save_payload()
