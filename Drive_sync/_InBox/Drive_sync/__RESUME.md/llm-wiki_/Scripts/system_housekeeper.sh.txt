#!/usr/bin/env bash
# ==============================================================================
# NOVÆXORPUS SYSTEM HOUSEKEEPER & ZERO-TRUST BOUNDARY SWEEPER
# ==============================================================================
set -euo pipefail


WORKSPACE_ROOT="${HOME}/novae-xorpus"
OBSIDIAN_VAULT="${HOME}/obsidian-vault-new"
BLUEPRINT_FILE="${WORKSPACE_ROOT}/master_blueprint.txt"
AESOP_REPO="${HOME}/repos/aesop-xi"


echo "=== [1/4] Checking Git Symlinks & Handoff Hooks ==="
if [ -d "$AESOP_REPO" ] && [ -d "$WORKSPACE_ROOT" ]; then
    mkdir -p "${WORKSPACE_ROOT}/projects/aesop-xi"
    ln -sfn "${AESOP_REPO}/RESUME.md" "${WORKSPACE_ROOT}/projects/aesop-xi/RESUME.md"
    ln -sfn "${AESOP_REPO}/CLAUDE.md" "${WORKSPACE_ROOT}/projects/aesop-xi/CLAUDE.md"
    
    HOOK_PATH="${AESOP_REPO}/.git/hooks/post-commit"
    if [ ! -f "$HOOK_PATH" ]; then
        cat << 'EOF' > "$HOOK_PATH"
#!/usr/bin/env bash
bash "${HOME}/novae-xorpus/tools/regenerate_masters.sh"
EOF
        chmod +x "$HOOK_PATH"
        echo "  --> Post-commit hook installed in aesop-xi."
    fi
    echo "  --> Git synchronization verified."
fi


echo "=== [2/4] Auditing Obsidian Vault Boundaries ==="
if [ -d "$OBSIDIAN_VAULT" ] && [ -f "$BLUEPRINT_FILE" ]; then
    UNINDEXED_FILES=0
    while IFS= read -r md_file; do
        fname=$(basename "$md_file")
        if ! grep -q "$fname" "$BLUEPRINT_FILE"; then
            echo "  [WARN] Unregistered asset: $fname (Context Leak Risk)"
            UNINDEXED_FILES=$((UNINDEXED_FILES + 1))
        fi
    done < <(find "$OBSIDIAN_VAULT" -type f -name "*.md")
    
    if [ "$UNINDEXED_FILES" -eq 0 ]; then
        echo "  --> Obsidian Vault matches Master Blueprint."
    else
        echo "  --> Notice: $UNINDEXED_FILES unindexed Markdown file(s) detected."
    fi
else
    echo "  --> Obsidian directory or master_blueprint.txt not found. Skipping hash check."
fi


echo "=== [3/4] Rebuilding Universal Manifest ==="
if [ -f "${WORKSPACE_ROOT}/tools/compile_manifest.py" ]; then
    python3 "${WORKSPACE_ROOT}/tools/compile_manifest.py"
else
    echo "  --> Manifest compiler script pending deployment."
fi


echo "=== [4/4] Housekeeper Sweep Complete ==="
