#!/usr/bin/env python3
"""
AESOP XI SYSTEM INTEGRITY AUDITOR
Enforces strict single-agent repository constraints across directory trees.
"""

import os
import sys

TARGET_DIRECTORIES = [
   "horizons-ui", "aesop-xi-memory", "termux-environment", 
   "file-management-system", "github-ci-cd", "skill-building-protocol", 
   "nova-claw-novus-agenti", "red-agent-auditor", "node-beta-jetson", 
   "node-gamma-rubik-pi", "global-documentation-vault"
]

def verify_structural_alignment():
   print("⏳ [LOCAL ENGINE] Initiating Zero-Trust File Structure Verification...")
   missing_assets = 0
   
   for folder in TARGET_DIRECTORIES:
       if not os.path.isdir(folder):
           print(f"❌ ERROR: Canonical workspace folder '{folder}' is missing.")
           missing_assets += 1
           continue
           
       # Global documentation vault holds references; other folders require structural manifests
       if folder == "global-documentation-vault":
           continue
           
       # Verify mandatory multi-tool components are anchored locally
       manifest_file = os.path.join(folder, "skill_manifest.json")
       wiki_file = os.path.join(folder, "llm_wiki.md")
       
       if not os.path.exists(manifest_file):
           print(f"⚠️ Layer anomaly found: Creating clean skill_manifest.json inside '{folder}'")
           with open(manifest_file, "w") as f:
               f.write('{\n  "layer_skills": [],\n  "mcp_routing_enabled": false\n}')
               
       if not os.path.exists(wiki_file):
           print(f"⚠️ Layer anomaly found: Initializing clear llm_wiki.md inside '{folder}'")
           with open(wiki_file, "w") as f:
               f.write(f"# Local Agent Wiki Meta: {folder.upper()}\n\n## Current Status\n* Isolated context bound active.")

   if missing_assets > 0:
       print(f"❌ Audit halted. Fix the {missing_assets} missing folders before processing storage states.")
       sys.exit(1)
   else:
       print("✅ SUCCESS: Ground-truth directory architecture matches master specifications.")

if __name__ == "__main__":
   verify_structural_alignment()
